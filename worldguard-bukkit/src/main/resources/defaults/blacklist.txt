#
# Черный список WorldGuard (Blacklist)
#
# Черный список позволяет блокировать использование предметов, блоков и выполнение действий.
# Вы выбираете набор «предметов для воздействия» и список «выполняемых действий».
#
###############################################################################
#
# Пример блокировки добычи и установки некоторых руд:
# [coal_ore,gold_ore,iron_ore]
# on-break=deny,log,kick
# on-place=deny,tell
#
# События, которые можно отследить:
# - on-break (когда блок этого типа пытаются сломать)
# - on-destroy-with (предмет/блок в руках игрока в момент разрушения чего-либо)
# - on-place (когда блок устанавливается)
# - on-use (использование предмета, например, огнива или ведра)
# - on-interact (взаимодействие с блоком: двери, сундуки и т. д.)
# - on-drop (предмет выбрасывается из инвентаря игрока)
# - on-acquire (предмет попадает в инвентарь игрока любым способом)
# - on-equip (предмет надевается в слоты брони)
#      ПРИМЕЧАНИЕ: on-equip работает слишком строго из-за ограничений Bukkit API
# - on-dispense (раздатчик собирается выбросить/использовать предмет)
#
# Действия (для событий):
# - deny (полный запрет, используется в режиме черного списка)
# - allow (разрешить, используется в режиме белого списка)
# - notify (уведомить админов с правом 'worldguard.notify')
# - log (записать в консоль/файл/базу данных)
# - tell (сообщить игроку, что это действие запрещено)
# - kick (кикнуть игрока)
# - ban (забанить игрока)
#
# Опции:
# - ignore-groups (список групп через запятую, на которых не влияет запрет)
# - ignore-perms (список прав через запятую, на которых не влияет запрет —
#     вы можете придумать свои собственные права!)
# - comment (ваша заметка, которая выводится при действиях 'log' и 'notify')
# - message (необязательное сообщение для игрока; %s заменяется на название предмета)
#
###############################################################################
#
# Для получения дополнительной информации посетите:
# https://worldguard.enginehub.org/en/latest/blacklist/
#
###############################################################################
#
# Ниже приведены примеры.
# ПОМНИТЕ: если строка начинается с #, она игнорируется.
#

#[lava_bucket]
#ignore-perms=my.own.madeup.permission
#ignore-groups=admins,mods
#on-use=deny,tell
